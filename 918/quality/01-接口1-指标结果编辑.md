package com.huawei.qua.ebg.outer.vo.samp;

import com.huawei.dispatchcenter.amsservice.validator.RegexpExpression;
import com.huawei.dispatchcenter.amsservice.validator.constraints.annotation.DataDictValidation;
import com.huawei.dispatchcenter.amsservice.validator.constraints.annotation.NotNullValidation;
import com.huawei.dispatchcenter.amsservice.validator.constraints.annotation.RegexpValidation;
import com.huawei.dispatchcenter.amsservice.validator.constraints.annotation.SizeValidation;
import com.huawei.qua.common.constants.DataConstants;

import jakarta.validation.Valid;
import lombok.Data;

import java.util.List;

/**
 * 采样待办任务---指标-编辑接口
 *
 * @author lWX1508535
 * @since 2026-06-08
 */
@Data
public class TodoTaskQuaIndicatorUpdateReqVo {
    /**
     * 采样待办任务id
     */
    @NotNullValidation
    private Long todoId;

    /**
     * 业务类型 (必填)
     * 1-问题单质检、2-重大问题质检
     */
    @NotNullValidation
    @DataDictValidation(dataCode = DataConstants.RULE_BUSI_TYPE)
    private String busiType;

    /**
     * 质检类型：CHECK_ITEM - 质检项、MONITORING_ITEM - 监控项
     */
    @NotNullValidation
    @DataDictValidation(dataCode = DataConstants.RULE_CHECK_TYPE)
    private String checkType;

    /**
     * 操作类型 (必填)
     * 初审指标编辑 / 复审指标编辑 / 关闭指标编辑 / 申诉指标编辑
     */
    @NotNullValidation
    @RegexpValidation(regexp = "^(1|2|3|4)$")
    private String operationType;

    /**
     * 质检指标 (必填)
     * varchar(64)，需符合规则1.15
     */
    @NotNullValidation
    @SizeValidation(max = 64, message = "质检指标长度不能超过64个字符")
    @RegexpValidation(regexp = RegexpExpression.REMARK_REGEX)
    private String indicator;

    /**
     * 备注/申诉原因
     * varchar(512)，需符合规则1.15
     */
    @SizeValidation(max = 512, message = "备注/申诉原因长度不能超过512个字符")
    @RegexpValidation(regexp = RegexpExpression.REMARK_REGEX)
    private String remarkOrAppealReason;

    /**
     * 考核责任人（CSE/CCR/PSE）(非必填)
     * varchar(64)，需符合规则1.15
     */
    @SizeValidation(max = 64, message = "考核责任人(CCR/PSE)长度不能超过64个字符")
    private List<@Valid @SizeValidation(max = 64)
    @RegexpValidation(regexp = RegexpExpression.REMARK_REGEX) String> assessor;

    /**
     * 判断结果 (非必填)
     * 0正常 / 1异常
     */
    @RegexpValidation(regexp = RegexpExpression.FLAG_REGEX)
    private String checkResult;

    /**
     * 是否申诉 (非必填)
     * 0是 / 1否
     */
    @RegexpValidation(regexp = RegexpExpression.FLAG_REGEX)
    private String isAppealed;

    /**
     * 文件id列表 (非必填)
     * long类型，用于关联附件表
     */
    @SizeValidation(max = 10, message = "附件数量不能超过10个")
    private List<Long> fileIds;

    /**
     * 是否涉及 (必填)
     * 0-涉及 / 1-不涉及
     */
    @NotNullValidation
    @RegexpValidation(regexp = RegexpExpression.FLAG_REGEX)
    private String isInvolve;
}
